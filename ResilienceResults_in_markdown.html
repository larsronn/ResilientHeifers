Time Budget Data from Two Farms

Material

Two commercial farms, one in Sweden (Farm A) and one in The Netherlands (Farm B). Data collected on 15 newly introduced heifers from each farm. The data also contains records from first lactating cows that have been in the barn for 100 days.

Functions

First I make some functions for computing 5-day means, standard deviations, and for plotting:

filter_summarise <- function(cow.data, hours.threshold=16) {
  den = 1
  test = cow.data$hours<=1
  if (all(test)) den=24
  cow.data$budget[which(cow.data$hours < hours.threshold/den)]=NA
  cow.data$Period = 1
  cow.data$Period[cow.data$day>20 & cow.data$day <26]= NA
  cow.data$Period[cow.data$day>25]= 2
  df = cow.data %>%
    group_by(Area, Period) %>%
    summarize(avg_time_per_day=mean(budget, na.rm=TRUE), proportion_NA = sum(is.na(budget))/length(budget)) %>%
    na.omit()
  print(df)
  return(cow.data)
}

Overall_GAMLSS <- function(cow.data, farm) {
  results = list()
  kk=0
  for (AREA in c("Cubicle", "Feed", "Alley")) {
  df1 <- cow.data[cow.data$Area==AREA,]
  df10 <- na.omit(df1)
  if (farm=="Farm A") mod.group<-gamlss(budget~cs(day)*factor(ind)*factor(Period) + 
                                          random(factor(date)) +n_intro + hours + trimming, 
                                      sigma.fo=~day + n_intro  + hours + trimming, 
                                      data=df10, control=gamlss.control(trace=FALSE),
                                        method=mixed(20,20))
  if (farm=="Farm B") mod.group<-gamlss(budget~cs(day)*factor(ind)*factor(Period) + 
                                          random(factor(date)) +n_intro + hours + factor(ind), 
                                      sigma.fo=~day + n_intro  + hours, 
                                      data=df10, control=gamlss.control(trace=FALSE),
                                        method=mixed(20,20))
  model_summary <- capture.output(summary(mod.group))
  print("######################################")
  print(AREA)
  print("Estimates of nuisance parameters not shown")
  test1 <- substr(model_summary, 1,7) == "n_intro"
  test2 <- test1 + 1*(substr(model_summary, 1,5) == "hours")
  test3 <- test2 + 1*(substr(model_summary, 1,8) == "trimming")
  test4 <- test3 + 1*(substr(model_summary, 1,3) == "day")
  test5 <- test4 + 1*(substr(model_summary, 1,2) == "Mu")
  test6 <- test5 + 1*(substr(model_summary, 1,5) == "Sigma")
  test7 <- test6 + 1*(substr(model_summary, 1,7) == "(Interc")
  #for (i in 7:(length(model_summary)-13)) cat(model_summary[i],"\n")
  for (i in which(test7==1)) cat(model_summary[i],"\n")
  kk= kk+1
  results[[kk]] = mod.group
  }

  return(results)
}

plot_GAMLSS_Results <- function(Results_GAMLSS, Heifers, farm) {
  kk = 0
  for (AREA in c("Cubicle", "Feed", "Alley")) {
    df1 <- Heifers[Heifers$Area==AREA,]
    df10 = na.omit(df1)
    kk = kk + 1
    mod.group = Results_GAMLSS[[kk]]
    y.all = NULL
    for (IND in unique(df1$ind)) {
      df11 = df10[df10$ind==IND,]
      df11$n_intro=0
      df11$hours=24
      if (farm=="Farm A") df11$trimming=0
      df12 =na.omit(df11)
      df12$date = df12$date[1]
      y=predict(mod.group, what="mu", newdata=df12, data=df10)
      y.all <- c(y.all, y)
    }
    df.all <- cbind(na.omit(df1), y.all)
    p <- ggplot(df.all, aes(day, y.all)) +
      geom_point() +
      geom_point(x=df.all$day, y=df.all$budget ,
               color="red") +
      facet_wrap(~ind) +
      ggtitle(paste(farm, AREA)) +
      ylab("Proportion of day") + 
      ylim(0,1)
    print(p)
  }
}
 
get.5dMean <- function(cow.data, area="Cubicle", remove5=FALSE) {
  m <- numeric(9)
  cow.data = cow.data[cow.data$Area==area,]
  for (i in 1:9) m[i] <- mean(cow.data[cow.data$day %in% (i*5-4):(i*5),]$budget, na.rm =TRUE)
  if (remove5) m <- m[-5]
  return(m)
}

Get.SDs <- function(cow.data, area="Cubicle", days=c(1,20)) {
  n = length(unique(cow.data$ind))
  vec.days <- days[1]:days[2]
  vec.sd <- numeric(n)
  for (id in 1:n) {
    vec.sd[id] = sd(cow.data[cow.data$Area==area & cow.data$ind==id & cow.data$day %in% vec.days,]$budget, na.rm=TRUE)
  }
  return(vec.sd)
}

Get.SDs_residuals <- function(cow.data, area="Cubicle", days=c(1,20)) {
  n = length(unique(cow.data$ind))
  vec.days <- days[1]:days[2]
  vec.sd <- numeric(n)
  for (id in 1:n) {
    y = cow.data[cow.data$Area==area & cow.data$ind==id & cow.data$day %in% vec.days,]$budget
    x = cow.data[cow.data$Area==area & cow.data$ind==id & cow.data$day %in% vec.days,]$day
    df1 = data.frame(x=x, y=y)
    mod1 = gamlss(y~cs(x), sigma.fo=~1, data=na.omit(df1), control=gamlss.control(trace=FALSE))
    res1 = mod1$mu.fv - df1$y
    vec.sd[id] = sd(res1, na.rm=TRUE)
  }
  return(vec.sd)
}

plot_smooth <- function(cow.data, AREA = "Cubicle", days=1:20, smooth.method="gam", main=NULL ) {
  df1 = cow.data[cow.data$Area==AREA,]
  df1 = df1[df1$day %in% days,]
  if (is.null(main)) main=paste("Days",days[1],"to",days[length(days)])
  ggplot(df1, aes(x=day, y=budget)) +
    geom_point() + 
    facet_wrap(~ind) + 
    geom_smooth(method=smooth.method) +
    ylab("Proportion of the day") +
    ggtitle(main)
}
